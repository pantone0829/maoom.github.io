<script>

/* ========================================
   MAOOM 5×5 PUZZLE
======================================== */

const N = 5;
const COUNT = N * N;
const EMPTY = COUNT - 1;

const board =
  document.getElementById("board");

const movesEl =
  document.getElementById("moves");

const timerEl =
  document.getElementById("timer");

const complete =
  document.getElementById("complete");


let state = [];

let moves = 0;

let locked = false;



/* ========================================
   TIMER
======================================== */

let startTime = null;

let timerInterval = null;

let elapsedTime = 0;


function formatTime(ms) {

  const totalSeconds =
    Math.floor(ms / 1000);

  const minutes =
    Math.floor(totalSeconds / 60);

  const seconds =
    totalSeconds % 60;

  return (
    String(minutes).padStart(2,"0")
    +
    ":"
    +
    String(seconds).padStart(2,"0")
  );
}


function updateTimer() {

  if(startTime === null) return;

  elapsedTime =
    Date.now() - startTime;

  timerEl.textContent =
    "TIME " + formatTime(elapsedTime);
}


function startTimer() {

  clearInterval(timerInterval);

  elapsedTime = 0;

  startTime = Date.now();

  timerEl.textContent =
    "TIME 00:00";

  timerInterval =
    setInterval(updateTimer,250);
}


function stopTimer() {

  if(startTime !== null) {

    elapsedTime =
      Date.now() - startTime;

  }

  clearInterval(timerInterval);

  timerInterval = null;

  startTime = null;

  timerEl.textContent =
    "TIME " + formatTime(elapsedTime);
}


function resetTimer() {

  clearInterval(timerInterval);

  timerInterval = null;

  startTime = null;

  elapsedTime = 0;

  timerEl.textContent =
    "TIME 00:00";
}



/* ========================================
   PUZZLE
======================================== */

function solved() {

  return Array.from(
    {length:COUNT},
    (_,i)=>i
  );

}


function rc(pos) {

  return [
    Math.floor(pos/N),
    pos%N
  ];

}


function adjacent(a,b) {

  const [ar,ac] = rc(a);

  const [br,bc] = rc(b);

  return (
    Math.abs(ar-br)
    +
    Math.abs(ac-bc)
    === 1
  );

}



/* ========================================
   TILE 생성
======================================== */

function build() {

  board.innerHTML = "";


  for(
    let id=0;
    id<EMPTY;
    id++
  ) {

    const tile =
      document.createElement("button");


    tile.className =
      "tile";


    tile.dataset.id =
      id;


    tile.type =
      "button";


    const row =
      Math.floor(id/N);


    const col =
      id%N;


    tile.style.backgroundPosition =
      `${col*25}% ${row*25}%`;


    tile.setAttribute(
      "aria-label",
      `퍼즐 조각 ${id+1}`
    );


    tile.addEventListener(
      "click",
      ()=>moveTile(id,true)
    );


    board.appendChild(tile);

  }

}



/* ========================================
   화면 표시
======================================== */

function render() {

  for(
    const tile
    of board.children
  ) {

    const id =
      Number(tile.dataset.id);


    const position =
      state.indexOf(id);


    const [row,col] =
      rc(position);


    tile.style.transform =
      `translate(
        ${col*100}%,
        ${row*100}%
      )`;

  }


  movesEl.textContent =
    "MOVE " + moves;

}



/* ========================================
   타일 이동
======================================== */

function moveTile(
  id,
  countMove
) {

  if(locked) return;


  const position =
    state.indexOf(id);


  const emptyPosition =
    state.indexOf(EMPTY);


  if(
    !adjacent(
      position,
      emptyPosition
    )
  ) return;


  [
    state[position],
    state[emptyPosition]
  ]
  =
  [
    state[emptyPosition],
    state[position]
  ];


  if(countMove) {

    moves++;

  }


  render();



  /* PUZZLE COMPLETE */

  if(
    countMove
    &&
    state.every(
      (value,index)=>
      value===index
    )
  ) {

    locked = true;

    stopTimer();

    complete
      .classList
      .add("show");

  }

}



/* ========================================
   SHUFFLE
======================================== */

function shuffle() {

  complete
    .classList
    .remove("show");


  locked = false;


  state =
    solved();


  moves = 0;


  let empty =
    EMPTY;


  let previous =
    -1;



  /*
  실제 가능한 이동을
  반복해서 섞기 때문에
  항상 풀 수 있는 퍼즐 생성
  */

  for(
    let i=0;
    i<400;
    i++
  ) {

    const [row,col] =
      rc(empty);


    const options = [];


    if(row>0)
      options.push(
        empty-N
      );


    if(row<N-1)
      options.push(
        empty+N
      );


    if(col>0)
      options.push(
        empty-1
      );


    if(col<N-1)
      options.push(
        empty+1
      );


    let choices =
      options.filter(
        x=>x!==previous
      );


    if(
      choices.length===0
    ) {

      choices =
        options;

    }


    const next =
      choices[
        Math.floor(
          Math.random()
          *
          choices.length
        )
      ];


    [
      state[empty],
      state[next]
    ]
    =
    [
      state[next],
      state[empty]
    ];


    previous =
      empty;


    empty =
      next;

  }



  /*
  혹시 완성 상태가 나오면
  다시 SHUFFLE
  */

  if(
    state.every(
      (value,index)=>
      value===index
    )
  ) {

    shuffle();

    return;

  }


  render();


  /* TIMER START */

  startTimer();

}



/* ========================================
   RESET
======================================== */

function reset() {

  complete
    .classList
    .remove("show");


  locked = false;


  state =
    solved();


  moves = 0;


  resetTimer();


  render();

}



/* ========================================
   MOBILE SWIPE
======================================== */

let startX = 0;

let startY = 0;


board.addEventListener(
  "touchstart",
  event => {

    const touch =
      event.touches[0];


    startX =
      touch.clientX;


    startY =
      touch.clientY;

  },
  {passive:true}
);



board.addEventListener(
  "touchend",
  event => {

    if(locked) return;


    const touch =
      event.changedTouches[0];


    const dx =
      touch.clientX-startX;


    const dy =
      touch.clientY-startY;



    if(
      Math.max(
        Math.abs(dx),
        Math.abs(dy)
      )
      <24
    ) {

      return;

    }



    const emptyPosition =
      state.indexOf(EMPTY);


    const [row,col] =
      rc(emptyPosition);


    let target =
      -1;



    /* LEFT / RIGHT */

    if(
      Math.abs(dx)
      >
      Math.abs(dy)
    ) {


      if(
        dx>0
        &&
        col>0
      ) {

        target =
          emptyPosition-1;

      }


      if(
        dx<0
        &&
        col<N-1
      ) {

        target =
          emptyPosition+1;

      }

    }


    /* UP / DOWN */

    else {


      if(
        dy>0
        &&
        row>0
      ) {

        target =
          emptyPosition-N;

      }


      if(
        dy<0
        &&
        row<N-1
      ) {

        target =
          emptyPosition+N;

      }

    }



    if(target>=0) {

      const id =
        state[target];


      if(id!==EMPTY) {

        moveTile(
          id,
          true
        );

      }

    }

  },
  {passive:true}
);



/* ========================================
   BUTTON
======================================== */

document
.getElementById("shuffle")
.addEventListener(
  "click",
  shuffle
);


document
.getElementById("reset")
.addEventListener(
  "click",
  reset
);



/* ========================================
   GAME START
======================================== */

build();

shuffle();


</script>
