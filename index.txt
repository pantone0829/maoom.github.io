  <script>
    const size = 5;
    const totalTiles = size * size;
    const board = document.getElementById("board");
    const movesEl = document.getElementById("moves");
    const timerEl = document.getElementById("timer");
    const winMessage = document.getElementById("winMessage");
    const shuffleBtn = document.getElementById("shuffleBtn");
    const resetBtn = document.getElementById("resetBtn");

    let tiles = [];
    let solvedTiles = [];
    let emptyIndex = totalTiles - 1;
    let moves = 0;
    let seconds = 0;
    let timerInterval = null;
    let gameStarted = false;

    function createSolvedState() {
      solvedTiles = [];
      for (let i = 0; i < totalTiles; i++) {
        solvedTiles.push(i);
      }
      tiles = [...solvedTiles];
      emptyIndex = totalTiles - 1;
    }

    function formatTime(sec) {
      const min = String(Math.floor(sec / 60)).padStart(2, "0");
      const s = String(sec % 60).padStart(2, "0");
      return `${min}:${s}`;
    }

    function startTimer() {
      if (timerInterval) return;
      timerInterval = setInterval(() => {
        seconds++;
        timerEl.textContent = formatTime(seconds);
      }, 1000);
    }

    function stopTimer() {
      clearInterval(timerInterval);
      timerInterval = null;
    }

    function resetGameStats() {
      moves = 0;
      seconds = 0;
      gameStarted = false;
      movesEl.textContent = moves;
      timerEl.textContent = "00:00";
      winMessage.textContent = "";
      stopTimer();
    }

    function renderBoard() {
      board.innerHTML = "";

      tiles.forEach((tileValue, index) => {
        const tile = document.createElement("div");

        if (tileValue === totalTiles - 1) {
          tile.className = "tile empty";
        } else {
          tile.className = "tile";
          tile.dataset.number = tileValue + 1;

          const row = Math.floor(tileValue / size);
          const col = tileValue % size;
          tile.style.backgroundPosition = `${(col / (size - 1)) * 100}% ${(row / (size - 1)) * 100}%`;

          tile.addEventListener("click", () => moveTile(index));
        }

        board.appendChild(tile);
      });
    }

    function getRowCol(index) {
      return {
        row: Math.floor(index / size),
        col: index % size
      };
    }

    function isAdjacent(index1, index2) {
      const a = getRowCol(index1);
      const b = getRowCol(index2);
      const rowDiff = Math.abs(a.row - b.row);
      const colDiff = Math.abs(a.col - b.col);
      return (rowDiff + colDiff === 1);
    }

    function moveTile(index) {
      if (!isAdjacent(index, emptyIndex)) return;

      if (!gameStarted) {
        gameStarted = true;
        startTimer();
      }

      [tiles[index], tiles[emptyIndex]] = [tiles[emptyIndex], tiles[index]];
      emptyIndex = index;
      moves++;
      movesEl.textContent = moves;
      renderBoard();

      if (checkSolved()) {
        stopTimer();
        winMessage.textContent = `완성! 총 ${moves}회 이동 / ${formatTime(seconds)} 소요`;
      }
    }

    function checkSolved() {
      return tiles.every((value, index) => value === solvedTiles[index]);
    }

    function getMovableIndices(emptyIdx) {
      const { row, col } = getRowCol(emptyIdx);
      const candidates = [];

      if (row > 0) candidates.push(emptyIdx - size);
      if (row < size - 1) candidates.push(emptyIdx + size);
      if (col > 0) candidates.push(emptyIdx - 1);
      if (col < size - 1) candidates.push(emptyIdx + 1);

      return candidates;
    }

    function shuffleBoard(steps = 300) {
      resetGameStats();
      createSolvedState();

      let lastIndex = -1;

      for (let i = 0; i < steps; i++) {
        let movable = getMovableIndices(emptyIndex);

        if (lastIndex !== -1) {
          movable = movable.filter(idx => idx !== lastIndex);
        }

        const randIndex = movable[Math.floor(Math.random() * movable.length)];
        [tiles[randIndex], tiles[emptyIndex]] = [tiles[emptyIndex], tiles[randIndex]];
        lastIndex = emptyIndex;
        emptyIndex = randIndex;
      }

      renderBoard();
    }

    function resetBoard() {
      resetGameStats();
      createSolvedState();
      renderBoard();
    }

    shuffleBtn.addEventListener("click", () => {
      shuffleBoard();
    });

    resetBtn.addEventListener("click", () => {
      resetBoard();
    });

    // 초기 실행
    createSolvedState();
    renderBoard();
  </script>
