startTime = null; elapsedTime = 0; timerEl.textContent = 'TIME 00:00'; } // =============================== // PUZZLE // =============================== function solved() { return Array.from( { length: COUNT }, (_, i) => i ); const p = state.indexOf(id); const e = state.indexOf(EMPTY); if (!adjacent(p, e)) return; [state[p], state[e]] = [state[e], state[p]]; if (countMove) { moves++; } render(); // 퍼즐 완성 if ( countMove && state.every((v, i) => v === i) ) { locked = true; [state[empty], state[next]] = [state[next], state[empty]]; prev = empty; empty = next; } // 우연히 완성 상태면 다시 섞기 if ( state.every((v, i) => v === i) ) { shuffle(); return; } render(); // ★ 여기서 타이머 시작 startTimer(); } // =============================== // RESET // =============================== function reset() { complete.classList.remove('show'); locked = false; state = solved(); moves = 0; resetTimer(); render(); } // =============================== // MOBILE SWIPE // =============================== let sx = 0; let sy = 0; board.addEventListener( 'touchstart', e => { const t = e.touches[0]; sx = t.clientX; sy = t.clientY; } else { if ( dy > 0 && er > 0 ) target = epos - N; if ( dy < 0 && er < N - 1 ) target = epos + N; } if (target >= 0) { const id = state[target]; if (id !== EMPTY) moveTile(id, true); } }, { passive: true } );
